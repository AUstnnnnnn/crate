# crate Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `crate`, a Textual TUI that downloads media (archive/audio-sample/video presets) via yt-dlp + ffmpeg, with in-TUI clip trimming.

**Architecture:** Small single-purpose Python modules behind explicit interfaces. Pure-logic modules (`timecode`, `library`, `config`) are built and tested first with no external deps. External tools (yt-dlp, ffmpeg, ffprobe) are wrapped behind thin interfaces so the TUI and tests depend on the interface, not the tool — keeping network/ffmpeg out of unit tests via mocks. The Textual `app` composes these into main / advanced-overlay / clip screens.

**Tech Stack:** Python 3, Textual, yt-dlp (as library), ffmpeg/ffprobe (subprocess), pytest, tomllib/tomli-w, macOS `afplay`.

## Global Constraints

- Python 3.11+ (uses stdlib `tomllib`). Run everything in a local `venv/`.
- yt-dlp used as a **library** (`import yt_dlp`), never shelled out.
- ffmpeg/ffprobe/afplay invoked via `subprocess`; no other external tools in v1.
- No network and no ffmpeg calls in unit tests — mock the interfaces.
- Package name `crate`, source under `src/crate/`, tests under `tests/`.
- Config path: `~/.config/crate/config.toml`. Library root default: `~/Desktop/Dev/sounds`.
- Presets: `a`=archive, `s`=sample (WAV default), `v`=video. Default preset configurable.
- Commit after every task (Conventional Commits). Each commit ends with the

```python
# src/crate/config.py
from __future__ import annotations

import tomllib
from dataclasses import dataclass, asdict
from pathlib import Path

import tomli_w

DEFAULT_CONFIG_PATH = Path("~/.config/crate/config.toml").expanduser()
_DEFAULT_LIBRARY_ROOT = "~/Desktop/Dev/sounds"


@dataclass
class Config:
    library_root: Path
    default_preset: str = "sample"
    sample_format: str = "wav"
    normalize: bool = False
    filename_template: str = "%(title)s"

    @classmethod
    def defaults(cls) -> "Config":
        return cls(library_root=Path(_DEFAULT_LIBRARY_ROOT).expanduser())


def load_config(path: Path | None = None) -> Config:
    path = path or DEFAULT_CONFIG_PATH
    if not path.exists():
        return Config.defaults()
    data = tomllib.loads(path.read_text())
    root = data.get("library_root", _DEFAULT_LIBRARY_ROOT)
    return Config(
        library_root=Path(root).expanduser(),
        default_preset=data.get("default_preset", "sample"),
        sample_format=data.get("sample_format", "wav"),
        normalize=data.get("normalize", False),
        filename_template=data.get("filename_template", "%(title)s"),
    )


def save_config(cfg: Config, path: Path | None = None) -> None:
    path = path or DEFAULT_CONFIG_PATH
    path.parent.mkdir(parents=True, exist_ok=True)
    data = asdict(cfg)
    data["library_root"] = str(cfg.library_root)
    path.write_text(tomli_w.dumps(data))
```

```python
# src/crate/timecode.py
from __future__ import annotations


def parse_timecode(s: str) -> float:
    s = s.strip()
    if not s:
        raise ValueError("empty timecode")
    parts = s.split(":")
    if len(parts) > 3:
        raise ValueError(f"too many segments: {s!r}")
    try:
        nums = [float(p) for p in parts]
    except ValueError as exc:
        raise ValueError(f"non-numeric timecode: {s!r}") from exc
    if any(n < 0 for n in nums):
        raise ValueError(f"negative timecode: {s!r}")
    # for multi-part forms, all but the leading segment are 0-59
    for n in nums[1:]:
        if n >= 60:
            raise ValueError(f"segment out of range: {s!r}")
    total = 0.0
    for n in nums:
        total = total * 60 + n
    return total


def format_timecode(seconds: float) -> str:
    seconds = int(round(seconds))
    h, rem = divmod(seconds, 3600)
    m, sec = divmod(rem, 60)
    if h:
        return f"{h}:{m:02d}:{sec:02d}"
    return f"{m}:{sec:02d}"


def format_slug(seconds: float) -> str:
    seconds = int(round(seconds))
    m, sec = divmod(seconds, 60)
    return f"{m}.{sec:02d}"
```

```python
# src/crate/library.py
from __future__ import annotations

import json
import re
from dataclasses import dataclass, asdict
from datetime import datetime, timezone
from pathlib import Path

from crate.config import Config
from crate.timecode import format_slug

_UNSAFE = re.compile(r'[/\\:\*\?"<>\|]')


@dataclass
class MediaMeta:
    source_url: str
    uploader: str
    title: str
    duration: float
    preset: str
    ext: str


def sanitize(name: str) -> str:
    cleaned = _UNSAFE.sub("_", name)
    return cleaned.strip().strip(".") or "untitled"


def resolve_path(cfg: Config, meta: MediaMeta) -> Path:
    root = cfg.library_root
    title = sanitize(meta.title)
    if meta.preset == "archive":
        return root / "archive" / sanitize(meta.uploader) / f"{title}.{meta.ext}"
    if meta.preset == "sample":
        return root / "samples" / f"{title}.{cfg.sample_format}"
    if meta.preset == "video":
        return root / "video" / f"{title}.mp4"
    raise ValueError(f"unknown preset: {meta.preset}")


def clip_path(cfg: Config, title: str, in_s: float, out_s: float) -> Path:
    title = sanitize(title)
    stem = f"{title}_{format_slug(in_s)}-{format_slug(out_s)}"
    return cfg.library_root / "samples" / "clips" / f"{stem}.{cfg.sample_format}"


def write_sidecar(path: Path, meta: MediaMeta) -> Path:
    side = path.with_suffix(path.suffix + ".json")
    data = asdict(meta)
    data["downloaded_at"] = datetime.now(timezone.utc).isoformat()
    side.parent.mkdir(parents=True, exist_ok=True)
    side.write_text(json.dumps(data, indent=2))
    return side
```

```python
# src/crate/probe.py
from __future__ import annotations

import json
import subprocess
from dataclasses import dataclass
from pathlib import Path


@dataclass
class ProbeResult:
    duration: float
    codec: str
    container: str


def parse_ffprobe_json(raw: str) -> ProbeResult:
    data = json.loads(raw)
    fmt = data.get("format", {})
    duration = float(fmt.get("duration", 0.0) or 0.0)
    container = fmt.get("format_name", "")
    codec = ""
    for stream in data.get("streams", []):
        if stream.get("codec_type") == "audio":
            codec = stream.get("codec_name", "")
            break
    if not codec and data.get("streams"):
        codec = data["streams"][0].get("codec_name", "")
    return ProbeResult(duration=duration, codec=codec, container=container)


def probe(path: Path, runner=subprocess.run) -> ProbeResult:
    argv = [
        "ffprobe", "-v", "quiet", "-print_format", "json",
        "-show_format", "-show_streams", str(path),
    ]
    proc = runner(argv, capture_output=True, text=True)
    return parse_ffprobe_json(proc.stdout)
```

```python
# src/crate/processor.py
from __future__ import annotations

import subprocess
from pathlib import Path


class ProcessorError(RuntimeError):
    pass


def build_clip_argv(src: Path, dst: Path, in_s: float, out_s: float) -> list[str]:
    # -ss after -i => sample-accurate (slower but exact) seek.
    return [
        "ffmpeg", "-hide_banner", "-y",
        "-i", str(src),
        "-ss", f"{in_s}", "-to", f"{out_s}",
        "-vn",
        str(dst),
    ]


def build_extract_argv(src: Path, dst: Path, normalize: bool) -> list[str]:
    argv = ["ffmpeg", "-hide_banner", "-y", "-i", str(src), "-vn"]
    if normalize:
        argv += ["-af", "loudnorm=I=-16:TP=-1.5:LRA=11"]
    argv.append(str(dst))
    return argv


def run_ffmpeg(argv: list[str], runner=subprocess.run) -> None:
    proc = runner(argv, capture_output=True, text=True)
    if proc.returncode != 0:
        raise ProcessorError(getattr(proc, "stderr", "") or "ffmpeg failed")
```

```python
from pathlib import Path

from crate.config import Config
```

```python
def build_ydl_opts(cfg: Config, preset: str, progress_hook) -> dict:
    staging = cfg.library_root / ".staging"
    opts: dict = {
        "format": PRESET_FORMATS[preset],
        "outtmpl": str(staging / "%(title)s.%(ext)s"),
        "progress_hooks": [progress_hook],
        "noplaylist": False,
        "quiet": True,
        "no_warnings": True,
    }
    if preset == "sample":
        opts["postprocessors"] = [{
            "key": "FFmpegExtractAudio",
            "preferredcodec": cfg.sample_format,
        }]
    elif preset == "video":
        opts["merge_output_format"] = "mp4"
    return opts


def info_to_meta(info: dict, preset: str) -> MediaMeta:
    return MediaMeta(
        source_url=info.get("webpage_url", info.get("original_url", "")),
        uploader=info.get("uploader", "unknown"),
        title=info.get("title", "untitled"),
        duration=float(info.get("duration") or 0.0),
        preset=preset,
        ext=info.get("ext", "bin"),
    )


def expand_entries(info: dict) -> list[dict]:
    if info.get("entries") is not None:
        return list(info["entries"])
    return [info]


class Downloader:
    """Integration wrapper around yt_dlp.YoutubeDL (not unit-tested)."""
```

```python
def progress_to_fraction(d: dict) -> float:
    if d.get("status") == "finished":
        return 1.0
    total = d.get("total_bytes") or d.get("total_bytes_estimate")
    got = d.get("downloaded_bytes")
    if total and got:
        return max(0.0, min(1.0, got / total))
    return 0.0
```

```python
    async def _real_download(self, item: QueueItem) -> None:
        import asyncio
        from crate.downloader import Downloader, info_to_meta

        item.status = "downloading"
        self._refresh_row(item)

        def hook(d):
            item.progress = progress_to_fraction(d)
            item.detail = d.get("_percent_str", "")

        try:
            dl = Downloader()
            staged = await asyncio.to_thread(
                dl.download, item.url, self.cfg, item.preset, hook
            )
            meta = info_to_meta(
                {"title": staged.stem, "ext": staged.suffix.lstrip("."),
                 "webpage_url": item.url}, item.preset)
            dest = resolve_path(self.cfg, meta)
            dest.parent.mkdir(parents=True, exist_ok=True)
            staged.replace(dest)
            write_sidecar(dest, meta)
            item.title = meta.title
            item.status = "done"
            item.detail = f"{dest.stat().st_size // 1024} KB"
        except Exception as exc:  # per-item isolation — never crash the app
            item.status = "error"
            item.detail = str(exc).splitlines()[0][:60]
        self._refresh_row(item)

    def _refresh_row(self, item: QueueItem) -> None:
        table = self.query_one("#queue", DataTable)
        glyph = {"queued": "⏸", "downloading": "⣾", "done": "✓", "error": "✗"}
        idx = self.items.index(item)
        row_key = list(table.rows)[idx]
        pct = f"{int(item.progress * 100)}%" if item.status == "downloading" else item.detail
        table.update_cell(row_key, table.ordered_columns[0].key, glyph[item.status])
        table.update_cell(row_key, table.ordered_columns[1].key, item.title)
        table.update_cell(row_key, table.ordered_columns[3].key, f"{item.status} {pct}")
```

```python
def validate_range(in_s: float, out_s: float, duration: float) -> None:
    if in_s < 0:
        raise ValueError("start before zero")
    if out_s <= in_s:
        raise ValueError("end must be after start")
    if duration and out_s > duration:
        raise ValueError("end past duration")

                in_s: float, out_s: float, runner=subprocess.run) -> Path:
    dst = clip_path(cfg, title, in_s, out_s)
    dst.parent.mkdir(parents=True, exist_ok=True)
    run_ffmpeg(build_clip_argv(source, dst, in_s, out_s), runner=runner)
    return dst
```
